cmake_minimum_required(VERSION 3.24)
project(MMViewer VERSION 0.1.0 LANGUAGES C CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(md4c STATIC vendor/md4c/md4c.c)
target_include_directories(md4c PUBLIC vendor/md4c)
add_library(mmcore STATIC src/native/core.cpp src/native/startup.cpp)
target_link_libraries(mmcore PUBLIC bcrypt)
target_include_directories(mmcore PUBLIC src/native)
add_library(mmdiagram STATIC src/native/diagram.cpp src/native/folder_icons.cpp)
target_include_directories(mmdiagram PUBLIC src/native)
target_link_libraries(mmdiagram PUBLIC gdiplus)
add_executable(MMViewer WIN32 src/native/main.cpp src/native/viewer.cpp src/native/app.rc)
target_link_libraries(MMViewer PRIVATE mmcore mmdiagram md4c comctl32 comdlg32 shell32 shlwapi ole32 dwmapi uxtheme gdiplus windowscodecs)
target_compile_definitions(MMViewer PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
foreach(target md4c mmcore mmdiagram MMViewer)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  if(MSVC)
    target_compile_options(${target} PRIVATE /utf-8 /W3 $<$<CONFIG:Release>:/Gy>)
  endif()
endforeach()
if(MSVC)
  target_link_options(MMViewer PRIVATE /MANIFEST:NO $<$<CONFIG:Release>:/OPT:REF> $<$<CONFIG:Release>:/OPT:ICF>)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE mmcore)
  add_executable(startup_tests tests/startup_tests.cpp)
  target_link_libraries(startup_tests PRIVATE mmcore)
  add_executable(diagram_tests tests/diagram_tests.cpp)
  target_link_libraries(diagram_tests PRIVATE mmdiagram)
  add_executable(viewer_tests tests/viewer_tests.cpp src/native/viewer.cpp)
  target_link_libraries(viewer_tests PRIVATE mmdiagram md4c gdiplus psapi uxtheme)
  add_executable(viewer_benchmark EXCLUDE_FROM_ALL tests/viewer_benchmark.cpp src/native/viewer.cpp)
  target_link_libraries(viewer_benchmark PRIVATE mmdiagram md4c gdiplus psapi uxtheme)
  target_compile_definitions(viewer_benchmark PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  if(MSVC)
    target_compile_options(viewer_benchmark PRIVATE /utf-8)
  endif()
  set(MMVIEWER_BASELINE_VIEWER "" CACHE FILEPATH "Optional earlier viewer.cpp for identical-workload comparison")
  if(MMVIEWER_BASELINE_VIEWER)
    add_executable(viewer_benchmark_baseline EXCLUDE_FROM_ALL tests/viewer_benchmark.cpp "${MMVIEWER_BASELINE_VIEWER}")
    target_link_libraries(viewer_benchmark_baseline PRIVATE mmdiagram md4c gdiplus psapi uxtheme)
    target_compile_definitions(viewer_benchmark_baseline PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    if(MSVC)
      target_compile_options(viewer_benchmark_baseline PRIVATE /utf-8)
    endif()
  endif()
  add_executable(host_tests tests/host_tests.cpp src/native/viewer.cpp src/native/app.rc)
  set_source_files_properties(tests/host_tests.cpp PROPERTIES
    OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/native/main.cpp")
  target_link_libraries(host_tests PRIVATE mmcore mmdiagram md4c comctl32 comdlg32 shell32 shlwapi ole32 dwmapi uxtheme gdiplus windowscodecs)
  if(MSVC)
    target_link_options(host_tests PRIVATE /MANIFEST:NO)
  endif()
  # Explicitly invoked endurance checks use isolated fixtures and session state.
  # Keep them out of ordinary builds/CTest because duration is configurable.
  set_source_files_properties(tests/soak_tests.cpp PROPERTIES
    OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/native/main.cpp;${CMAKE_CURRENT_SOURCE_DIR}/tests/host_tests.cpp")
  # Separate output permits a heap-instrumented rerun while an older soak
  # executable remains alive; the workload and source stay identical.
  foreach(soakTarget soak_tests soak_heap_tests)
    add_executable(${soakTarget} EXCLUDE_FROM_ALL tests/soak_tests.cpp src/native/viewer.cpp src/native/app.rc)
    target_link_libraries(${soakTarget} PRIVATE mmcore mmdiagram md4c comctl32 comdlg32 shell32 shlwapi ole32 dwmapi uxtheme gdiplus windowscodecs psapi)
    target_compile_definitions(${soakTarget} PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    if(MSVC)
      target_compile_options(${soakTarget} PRIVATE /utf-8 /UNDEBUG)
      target_link_options(${soakTarget} PRIVATE /MANIFEST:NO)
    endif()
  endforeach()
  foreach(probe lifetime_probe async_probe image_lifecycle_probe expansion_tests tree_probe)
    if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/${probe}.cpp")
      continue()
    endif()
    add_executable(${probe} tests/${probe}.cpp src/native/viewer.cpp src/native/app.rc)
    set_source_files_properties(tests/${probe}.cpp PROPERTIES
      OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/native/main.cpp;${CMAKE_CURRENT_SOURCE_DIR}/tests/host_tests.cpp")
    target_link_libraries(${probe} PRIVATE mmcore mmdiagram md4c comctl32 comdlg32 shell32 shlwapi ole32 dwmapi uxtheme gdiplus windowscodecs psapi)
    target_compile_definitions(${probe} PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    if(MSVC)
      target_compile_options(${probe} PRIVATE /utf-8 /UNDEBUG)
      target_link_options(${probe} PRIVATE /MANIFEST:NO)
    endif()
  endforeach()
  foreach(target core_tests startup_tests diagram_tests viewer_tests host_tests)
    target_compile_definitions(${target} PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    if(MSVC)
      target_compile_options(${target} PRIVATE /utf-8 /UNDEBUG)
    endif()
  endforeach()
  add_test(NAME core COMMAND core_tests)
  add_test(NAME startup COMMAND startup_tests)
  add_test(NAME diagram COMMAND diagram_tests)
  add_test(NAME viewer COMMAND viewer_tests)
  add_test(NAME host COMMAND host_tests)
  add_test(NAME host_tree_scroll COMMAND host_tests --tree-scroll-regression)
  set_tests_properties(core startup diagram viewer host PROPERTIES TIMEOUT 120)
  # The native tree regression completes in about 100 s on an idle machine and
  # only needs a bound that distinguishes a hang; a loaded machine doubles it.
  set_tests_properties(host_tree_scroll PROPERTIES TIMEOUT 300)
  add_test(NAME session_bulk COMMAND lifetime_probe)
  add_test(NAME encoding_reentry COMMAND async_probe --encoding-reentry)
  add_test(NAME watch_recovery COMMAND async_probe)
  add_test(NAME image_lifecycle COMMAND image_lifecycle_probe)
  add_test(NAME expansion_history COMMAND expansion_tests)
  set_tests_properties(session_bulk encoding_reentry watch_recovery image_lifecycle expansion_history PROPERTIES TIMEOUT 120)
endif()
